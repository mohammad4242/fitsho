    85	def generate_program(
    86	    request: ProgramGenerationRequest,
    87	    exercise_catalog: list[ExerciseCandidate] | tuple[ExerciseCandidate, ...],
    88	    ruleset: ProgramRuleset,
    89	    *,
    90	    reference_templates: tuple[TemplateReference, ...] = (),
    91	) -> ProgramGenerationResult:
    92	    training_day_error = _training_day_error(request, ruleset)
    93	    if training_day_error is not None:
    94	        return ProgramGenerationResult(
    95	            program=None,
    96	            error_code=GenerationErrorCode.UNSUPPORTED_RESISTANCE_TRAINING_DAYS,
    97	            errors=(training_day_error,),
    98	        )
    99	    request = request.model_copy(
   100	        update={
   101	            "body_analysis_influence": applicable_body_analysis_influence(
   102	                request.body_analysis_influence, ruleset
   103	            )
   104	        }
   105	    )
   106	    normalized = normalize_request(request, ruleset)
   107	    safety = screen_safety(normalized)
   108	    if safety.status not in {SafetyStatus.CLEAR, SafetyStatus.CLEAR_WITH_MODIFICATIONS}:
   109	        return ProgramGenerationResult(
   110	            program=None,
   111	            error_code=GenerationErrorCode.PROGRAM_REJECTED_SAFETY_STATUS,
   112	            errors=safety.reason_codes,
   113	            safety_status=safety.status,
   114	        )
   115	    eligibility = filter_eligible_exercises(normalized, exercise_catalog)
   116	    session_capacity = build_session_capacity(
   117	        normalized,
   118	        eligibility.eligible,
   119	        ruleset,
   120	    )
   121	    template_selection = select_template_reference_result(
   122	        normalized,
   123	        eligibility.eligible,
   124	        reference_templates,
   125	        ruleset,
   126	        session_capacity=session_capacity,
   127	    )
   128	    template_selection_trace = (template_selection.decision_trace(),)
   129	    if not eligibility.eligible:
   130	        missing_equipment = any(
   131	            "EXERCISE_REJECTED_MISSING_EQUIPMENT" in item.reason_codes
   132	            for item in eligibility.rejected
   133	        )
   134	        return ProgramGenerationResult(
   135	            program=None,
   136	            error_code=(
   137	                GenerationErrorCode.NO_AVAILABLE_EQUIPMENT_MATCH
   138	                if missing_equipment
   139	                else GenerationErrorCode.INSUFFICIENT_ELIGIBLE_EXERCISES
   140	            ),
   141	            errors=("NO_ELIGIBLE_EXERCISES",),
   142	            safety_status=safety.status,
   143	            rejected_candidates=eligibility.rejected,
   144	            decision_trace=template_selection_trace,
   145	        )
   146	    previous_volume = derive_previous_volume_baseline(normalized.source.recent_training_history)
   147	    rejected_by_id = {item.exercise_id: item.reason_codes for item in eligibility.rejected}
   148	    rejected_slot_candidates = tuple(
   149	        (candidate, rejected_by_id[candidate.id])
   150	        for candidate in exercise_catalog
   151	        if candidate.id in rejected_by_id
   152	    )
   153	    template_rejection_trace: tuple[dict[str, object], ...] = template_selection_trace
   154	    for ranking in template_selection.candidates:
   155	        reference = ranking.template
   156	        try:
   157	            reference_build = build_template_sessions(
   158	                normalized,
   159	                reference,
   160	                eligibility.eligible,
   161	                ruleset,
   162	                session_capacity=session_capacity,
   163	                exercise_catalog=exercise_catalog,
   164	            )
   165	            reference_result = _reference_program(
   166	                request,
   167	                normalized,
   168	                safety.status,
   169	                safety.reason_codes,
   170	                eligibility.rejected,
   171	                eligibility.eligible,
   172	                eligibility.cardio_eligible,
   173	                reference,
   174	                reference_build,
   175	                ruleset,
   176	                previous_volume=previous_volume,
   177	                session_capacity=session_capacity,
   178	                template_selection_trace=template_rejection_trace,
   179	            )
   180	            if reference_result.is_success:
   181	                return _append_successful_template_attempt(
   182	                    reference_result,
   183	                    _template_attempt_trace(ranking, status="succeeded"),
   184	                )
   185	            rejection_category = _template_rejection_category(reference_result.errors)
   186	            rejection = {
   187	                "stage": "template_reference",
   188	                "rank": ranking.rank,
   189	                "selected": reference.slug,
   190	                "status": "rejected",
   191	                "hard_eligibility": (
   192	                    "days",
   193	                    "training_level",
   194	                    "core_slots_resolvable",
   195	                ),
   196	                "goal_used_for_exclusion": False,
   197	                "score": ranking.decision_trace()["score"],
   198	                "feasibility": ranking.feasibility.decision_trace(),
   199	                "rejection_category": rejection_category,
   200	                "reason_codes": reference_result.errors,
   201	                "decision_trace": reference_result.decision_trace[len(template_rejection_trace) :],
   202	            }
   203	            template_rejection_trace += (
   204	                rejection,
   205	                _template_attempt_trace(
   206	                    ranking,
   207	                    status="rejected",
   208	                    rejection_category=rejection_category,
   209	                    reason_codes=reference_result.errors,
   210	                ),
   211	            )
   212	        except TemplateConstructionError as exc:
   213	            reason_codes = ("INITIAL_TEMPLATE_REJECTED_UNFILLABLE", *exc.reason_codes)
   214	            rejection_category = _template_rejection_category(reason_codes)
   215	            rejection = {
   216	                "stage": "template_reference",
   217	                "rank": ranking.rank,
   218	                "selected": reference.slug,
   219	                "status": "rejected",
   220	                "hard_eligibility": (
   221	                    "days",
   222	                    "training_level",
   223	                    "core_slots_resolvable",
   224	                ),
   225	                "goal_used_for_exclusion": False,
   226	                "score": ranking.decision_trace()["score"],
   227	                "feasibility": ranking.feasibility.decision_trace(),
   228	                "rejection_category": rejection_category,
   229	                "reason_codes": reason_codes,
   230	            }
   231	            template_rejection_trace += (
   232	                rejection,
   233	                _template_attempt_trace(
   234	                    ranking,
   235	                    status="rejected",
   236	                    rejection_category=rejection_category,
   237	                    reason_codes=reason_codes,
   238	                ),
   239	            )
   240	    if template_selection.candidates:
   241	        template_rejection_trace += (
   242	            {
   243	                "stage": "template_recovery",
   244	                "status": "exhausted",
   245	                "attempted_count": len(template_selection.candidates),
   246	                "candidate_count": len(template_selection.candidates),
   247	                "reason_codes": ("TEMPLATE_ALTERNATIVES_EXHAUSTED",),
   248	            },
   249	        )
   250	    requested_days = normalized.resistance_training_days
   251	    ranked_splits = rank_split_candidates(
   252	        normalized,
   253	        ruleset,
   254	        exercises=eligibility.eligible,
   255	        session_capacity=session_capacity,
   256	    )
   257	    exact_day_splits = tuple(
   258	        candidate for candidate in ranked_splits if len(candidate.day_focuses) == requested_days
   259	    )
   260	    if not exact_day_splits:
   261	        collected_errors: list[str] = [
   262	            "REQUESTED_TRAINING_DAYS_UNSATISFIED",
   263	            "NO_EXACT_DAY_SPLIT_AVAILABLE",
   264	        ]
   265	        rejected_splits: list[dict[str, object]] = []
   266	    else:
   267	        rejected_splits = []
   268	        collected_errors = []
   269	    for attempt_index, candidate in enumerate(exact_day_splits):
   270	        split = candidate
   271	        if attempt_index:
   272	            split = replace(
   273	                candidate,
   274	                reason_codes=candidate.reason_codes
   275	                + ("SPLIT_FALLBACK_AFTER_CONSTRUCTION_FAILURE",),
   276	            )
   277	        result = _program_for_split(
   278	            request,
   279	            normalized,
   280	            safety.status,
   281	            safety.reason_codes,
   282	            eligibility.rejected,
   283	            eligibility.eligible,
   284	            eligibility.cardio_eligible,
   285	            split,
   286	            ruleset,
   287	            previous_volume=previous_volume,
   288	            session_capacity=session_capacity,
   289	            rejected_splits=tuple(rejected_splits),
   290	            template_rejection_trace=template_rejection_trace,
   291	            rejected_slot_candidates=rejected_slot_candidates,
   292	        )
   293	        if result.is_success:
   294	            return result
   295	        collected_errors.extend(result.errors)
   296	        rejected_attempt: dict[str, object] = {
   297	            "split": split.split_type.value,
   298	            "day_focuses": split.day_focuses,
   299	            "status": "rejected",
   300	            "reason_codes": result.errors,
   301	        }
   302	        if result.decision_trace:
   303	            rejected_attempt["decision_trace"] = result.decision_trace
   304	        rejected_splits.append(rejected_attempt)
   305	
   306	    weekdays_fallback = (
   307	        exact_day_splits[0].weekdays if exact_day_splits else tuple(range(requested_days))
   308	    )
   309	    dynamic_splits = rank_availability_aware_fallbacks(
   310	        normalized,
   311	        eligibility.eligible,
   312	        ruleset,
   313	        weekdays=weekdays_fallback,
   314	        excluded_layouts=frozenset(candidate.day_focuses for candidate in exact_day_splits),
   315	        session_capacity=session_capacity,
   316	    )
   317	    for fallback_split in dynamic_splits:
   318	        result = _program_for_split(
   319	            request,
   320	            normalized,
   321	            safety.status,
   322	            safety.reason_codes,
   323	            eligibility.rejected,

   326	            fallback_split,
   327	            ruleset,
   328	            previous_volume=previous_volume,
   329	            session_capacity=session_capacity,
   330	            rejected_splits=tuple(rejected_splits),
   331	            template_rejection_trace=template_rejection_trace,
   332	            rejected_slot_candidates=rejected_slot_candidates,
   333	        )
   334	        if result.is_success:
   335	            return result
   336	        collected_errors.extend(result.errors)
   337	        rejected_splits.append(
   338	            {
   339	                "split": fallback_split.split_type.value,
   340	                "day_focuses": fallback_split.day_focuses,
   341	                "status": "rejected",
   342	                "reason_codes": result.errors,
   343	                "decision_trace": result.decision_trace,
   344	            }
   345	        )
   346	    print('BEFORE TUPLE: ', collected_errors)
   347	    errors = (
   348	        "PROGRAM_CONSTRUCTION_ALTERNATIVES_EXHAUSTED",
   349	        "EXACT_DAY_SPLIT_ALTERNATIVES_EXHAUSTED",
   350	        "REQUESTED_TRAINING_DAYS_UNSATISFIED",
   351	        *tuple(dict.fromkeys(collected_errors)),
   352	    )
   353	    print('AFTER TUPLE: ', errors)
   354	    print('RETURNING RESULT WITH ERRORS:', errors)
   355	    print('RETURNING FOR SEED:', getattr(request, 'seed', 'no seed'))
   356	    return ProgramGenerationResult(
   357	        program=None,
   358	        error_code=GenerationErrorCode.UNSATISFIED_CONSTRAINT,
   359	        errors=errors,
   360	        safety_status=safety.status,
   361	        rejected_candidates=eligibility.rejected,
   362	        decision_trace=(
   363	            {
   364	                "stage": "construction_recovery",
   365	                "status": "exhausted",
   366	                "attempts": tuple(rejected_splits),
   367	                "reason_codes": errors,
   368	            },
   369	        ),

   373	def _template_rejection_category(reason_codes: tuple[str, ...]) -> str:
   374	    joined = " ".join(reason_codes)
   375	    if any(marker in joined for marker in ("CORE_SLOT", "REQUIRED_CORE")):
   376	        return "CORE_SLOT_UNRESOLVED"
   377	    if any(marker in joined for marker in ("SAFETY", "EQUIPMENT", "NO_ELIGIBLE")):
   378	        return "SAFETY_EQUIPMENT_INCOMPATIBILITY"
   379	    if "PRIORITY_HARD_MINIMUM" in joined:
   380	        return "HARD_PRIORITY_MINIMUM_FAILURE"
   381	    if any(marker in joined for marker in ("DURATION", "RECOVERY")):
   382	        return "DURATION_RECOVERY_HARD_IMPOSSIBILITY"
   383	    if any(
   384	        marker in joined
   385	        for marker in (
   386	            "VALIDATION",
   387	            "REQUIRED_MOVEMENT",
   388	            "WEEKLY_MUSCLE_VOLUME",
   389	            "PER_SESSION_MUSCLE_VOLUME",
   390	            "REQUESTED_TRAINING_DAYS",
   391	        )
   392	    ):
   393	        return "VALIDATION_FAILURE"
   394	    return "ADAPTATION_EXHAUSTED"

   417	def _append_successful_template_attempt(
   418	    result: ProgramGenerationResult,
   419	    attempt_trace: dict[str, object],
   420	) -> ProgramGenerationResult:
   421	    if result.program is None:
   422	        return result
   423	    coach_index = next(
   424	        (
   425	            index
   426	            for index, entry in enumerate(result.program.decision_trace)
   427	            if entry.get("stage") == "final_construction"
   428	        ),
   429	        len(result.program.decision_trace),
   430	    )
   431	    final_trace = (
   432	        result.program.decision_trace[:coach_index]
   433	        + (attempt_trace,)
   434	        + result.program.decision_trace[coach_index:]
   435	    )
   436	    report = replace(result.program.validation_report, decision_trace=final_trace)
   437	    program = replace(
   438	        result.program,
   439	        validation_report=report,
   440	        decision_trace=final_trace,
   441	    )
   442	    return replace(result, program=program, decision_trace=final_trace)

   445	def _training_day_error(
   446	    request: ProgramGenerationRequest,
   447	    ruleset: ProgramRuleset,
   448	) -> str | None:
   449	    requested_days = request.available_training_days
   450	    if requested_days > SUPPORTED_RESISTANCE_TRAINING_DAYS[-1]:
   451	        return "UNSUPPORTED_RESISTANCE_TRAINING_DAYS"
   452	    if requested_days < SUPPORTED_RESISTANCE_TRAINING_DAYS[0]:
   453	        return None
   454	    if requested_days > ruleset.max_resistance_days:
   455	        return "UNSUPPORTED_RESISTANCE_TRAINING_DAYS"
   456	    experience_level = ExperienceLevel(request.training_experience.value)
   457	    if (
   458	        resistance_training_day_status(experience_level, requested_days)
   459	        is ResistanceTrainingDayStatus.UNSUPPORTED
   460	    ):
   461	        return "UNSUPPORTED_RESISTANCE_TRAINING_DAYS"
   462	    return None

   465	def _rejected_split_recovery_reasons(
   466	    rejected_splits: tuple[dict[str, object], ...],
   467	) -> tuple[str, ...]:
   468	    reasons: list[str] = []
   469	    for rejected_split in rejected_splits:
   470	        decision_trace = rejected_split.get("decision_trace")
   471	        if not isinstance(decision_trace, tuple):
   472	            continue
   473	        for entry in decision_trace:
   474	            if not isinstance(entry, dict) or entry.get("stage") != "construction_recovery":
   475	                continue
   476	            reason_codes = entry.get("reason_codes")
   477	            if isinstance(reason_codes, tuple):
   478	                reasons.extend(code for code in reason_codes if isinstance(code, str))
   479	    return tuple(dict.fromkeys(reasons))

   482	def _program_for_split(
   483	    request: ProgramGenerationRequest,
   484	    normalized: NormalizedProgramRequest,
   485	    safety_status: SafetyStatus,
   486	    safety_reasons: tuple[str, ...],
   487	    rejected: tuple[RejectedCandidate, ...],
   488	    eligible: tuple[ExerciseCandidate, ...],
   489	    cardio_eligible: tuple[ExerciseCandidate, ...],
   490	    split: SplitPlan,
   491	    ruleset: ProgramRuleset,
   492	    *,
   493	    previous_volume: PreviousVolumeBaseline,
   494	    session_capacity: SessionCapacity,
   495	    rejected_splits: tuple[dict[str, object], ...],
   496	    template_rejection_trace: tuple[dict[str, object], ...],
   497	    rejected_slot_candidates: tuple[tuple[ExerciseCandidate, tuple[str, ...]], ...],
   498	) -> ProgramGenerationResult:
   499	    volume = plan_weekly_volume(
   500	        normalized,
   501	        split,
   502	        ruleset,
   503	        previous_volume=previous_volume,
   504	        session_capacity=session_capacity,
   505	    )
   506	    try:
   507	        drafts = build_sessions(
   508	            normalized,
   509	            split,
   510	            volume,
   511	            eligible,
   512	            ruleset,
   513	            rejected_slot_candidates=rejected_slot_candidates,
   514	            session_capacity=session_capacity,
   515	        )
   516	    except SessionConstructionError as exc:
   517	        construction_trace: tuple[dict[str, object], ...] = template_rejection_trace + (
   518	            {
   519	                "stage": "session_construction",
   520	                "status": "rejected",
   521	                "split": split.split_type.value,
   522	                "day_index": exc.day_index,
   523	                "focus": exc.focus,
   524	                "required_patterns": exc.patterns,
   525	                "required_target_muscle": exc.target_muscle,
   526	                "candidate_rejection_reasons": exc.rejection_reasons,
   527	                "reason_codes": exc.reason_codes,
   528	            },
   529	        )
   530	        return ProgramGenerationResult(
   531	            program=None,
   532	            error_code=GenerationErrorCode.NO_SAFE_EXERCISE_FOR_PATTERN,
   533	            errors=exc.reason_codes,
   534	            safety_status=safety_status,
   535	            rejected_candidates=rejected,
   536	            decision_trace=construction_trace,
   537	        )
   538	    substitution_decisions = [
   539	        decision for draft in drafts for decision in draft.substitution_decisions
   540	    ]
   541	    days = prescribe_sessions(
   542	        normalized,
   543	        drafts,
   544	        volume,
   545	        ruleset,
   546	    )
   547	    days, repair_reasons = repair_weekly_volume(
   548	        days,
   549	        normalized,
   550	        volume,
   551	        ruleset,
   552	        candidates=eligible,
   553	        substitution_decisions=substitution_decisions,
   554	    )
   555	    days_before_duration_repair = days
   556	    days, duration_repair_reasons = repair_session_durations(
   557	        days,
   558	        normalized,
   559	        eligible,
   560	        ruleset,
   561	        volume=volume,
   562	        session_capacity=session_capacity,
   563	    )
   564	    split, days, recovery_repair_reasons = repair_recovery_weekdays(split, days, ruleset)
   565	    days = finalize_session_structure(days, normalized, ruleset)
   566	    day_count_errors = _day_count_errors(
   567	        len(days), normalized.resistance_training_days, stage="dynamic_construction"
   568	    )
   569	    if day_count_errors:
   570	        return ProgramGenerationResult(
   571	            program=None,
   572	            error_code=GenerationErrorCode.UNSATISFIED_CONSTRAINT,
   573	            errors=day_count_errors,
   574	            safety_status=safety_status,
   575	            rejected_candidates=rejected,
   576	            decision_trace=template_rejection_trace
   577	            + (
   578	                {
   579	                    "stage": "day_count_invariant",
   580	                    "status": "rejected",
   581	                    "expected_days": normalized.resistance_training_days,
   582	                    "actual_days": len(days),
   583	                    "split": split.split_type.value,
   584	                    "reason_codes": day_count_errors,
   585	                },
   586	            ),
   587	        )
   588	    effective_volume = calculate_effective_volume(
   589	        (item for day in days for item in day.exercises),
   590	        ruleset,
   591	    )
   592	    direct = Counter(effective_volume.direct_sets_by_muscle)
   593	    relaxed_groups = tuple(
   594	        dict.fromkeys(group for draft in drafts for group in draft.relaxed_required_pattern_groups)
   595	    )
   596	    relaxed_slots = tuple(
   597	        dict.fromkeys(
   598	            (group, target)
   599	            for draft in drafts
   600	            for group, target in zip(
   601	                draft.relaxed_required_pattern_groups,
   602	                draft.relaxed_required_target_muscles,
   603	                strict=True,
   604	            )
   605	        )
   606	    )
   607	    substitution_metrics = aggregate_substitution_observability(substitution_decisions)
   608	    metrics = {
   609	        **_volume_metrics(
   610	            days,
   611	            volume,
   612	            previous_volume,
   613	            ruleset,
   614	            request=normalized,
   615	            relaxed_required_pattern_groups=relaxed_groups,
   616	            relaxed_required_slots=relaxed_slots,
   617	            repair_reason_codes=tuple(dict.fromkeys((*repair_reasons, *duration_repair_reasons))),
   618	        ),
   619	        **substitution_metrics,
   620	    }
   621	    body_trace = body_analysis_trace(normalized, ruleset)
   622	    session_reasons = tuple(
   623	        dict.fromkeys(reason for draft in drafts for reason in draft.reason_codes)
   624	    )
   625	    recovery_reasons = tuple(
   626	        dict.fromkeys(
   627	            (
   628	                *(("SPLIT_FALLBACK_AFTER_CONSTRUCTION_FAILURE",) if rejected_splits else ()),
   629	                *_rejected_split_recovery_reasons(rejected_splits),
   630	                *session_reasons,
   631	                *recovery_repair_reasons,
   632	                *duration_repair_reasons,
   633	            )
   634	        )
   635	    )
   636	    trace: tuple[dict[str, object], ...] = template_rejection_trace + (
   637	        {"stage": "normalization", "assumptions": normalized.assumptions},
   638	        {"stage": "safety", "status": safety_status.value, "reasons": safety_reasons},
   639	        {
   640	            "stage": "eligibility",
   641	            "eligible_count": len(eligible),
   642	            "rejected_count": len(rejected),
   643	        },
   644	        _duration_capacity_trace(session_capacity),
   645	        *((body_trace,) if body_trace is not None else ()),
   646	        {
   647	            "stage": "construction_recovery",
   648	            "status": "recovered" if recovery_reasons else "not_required",
   649	            "selected_split": split.split_type.value,
   650	            "rejected_splits": rejected_splits,
   651	            "reason_codes": recovery_reasons,
   652	            "session_reasons": tuple(
   653	                {
   654	                    "day_index": draft.day_index,
   655	                    "focus": draft.focus,
   656	                    "reason_codes": draft.reason_codes,
   657	                }
   658	                for draft in drafts
   659	                if draft.reason_codes
   660	            ),
   661	        },
   662	        {
   663	            "stage": "day_count_invariant",
   664	            "status": "satisfied",
   665	            "expected_days": normalized.resistance_training_days,
   666	            "actual_days": len(days),
   667	            "reason_codes": ("REQUESTED_TRAINING_DAYS_SATISFIED",),
   668	        },
   669	        {"stage": "split", "selected": split.split_type.value, "reasons": split.reason_codes},
   670	        _volume_decision_trace(volume, previous_volume, normalized, ruleset),
   671	        {
   672	            "stage": "volume_repair",
   673	            "reasons": tuple(dict.fromkeys((*repair_reasons, *recovery_repair_reasons))),
   674	            "weekly_direct_sets": dict(direct),
   675	            "weekly_effective_sets": effective_volume.effective_sets_by_muscle,
   676	        },
   677	        _duration_repair_trace(
   678	            session_capacity,
   679	            days_before_duration_repair,
   680	            days,
   681	            duration_repair_reasons,
   682	        ),
   683	        {
   684	            "stage": "session_structure",
   685	            "status": "finalized",
   686	            "reason_codes": ("FINAL_SESSION_SEQUENCE_APPLIED",),
   687	        },
   688	        {
   689	            "stage": "substitution_observability",
   690	            "metrics": substitution_metrics,
   691	            "decisions": substitution_decision_summaries(substitution_decisions),
   692	        },
   693	    )
   694	    empty_report = ValidationReport(
   695	        errors=(),
   696	        warnings=(),
   697	        assumptions=normalized.assumptions,
   698	        metrics=metrics,
   699	        decision_trace=trace,
   700	    )
   701	    days = add_cardio(normalized, days, cardio_eligible, ruleset)
   702	    program = WorkoutProgram(
   703	        user_profile_snapshot=request.model_dump(mode="json"),
   704	        engine_version=ruleset.engine_version,
   705	        ruleset_version=ruleset.version,
   706	        seed=normalized.seed,
   707	        primary_goal=normalized.primary_goal,
   708	        secondary_goal=request.secondary_goal_optional,
   709	        training_status=normalized.training_status,
   710	        safety_status=safety_status,
   711	        assumptions=normalized.assumptions,
   712	        warnings=(),
   713	        duration_weeks=request.program_duration_weeks,
   714	        split=split,
   715	        weekly_schedule=days,
   716	        progression_policy={
   717	            **double_progression_policy(ruleset),
   718	            "deload": deload_policy(ruleset),
   719	        },
   720	        validation_report=empty_report,
   721	        aggregate_metrics=metrics,
   722	        decision_trace=trace,
   723	        body_analysis_provenance=body_analysis_provenance(normalized),
   724	    )
   725	    report = validate_program(program, request, ruleset)
   726	    if not report.is_valid:
   727	        return ProgramGenerationResult(
   728	            program=None,
   729	            error_code=GenerationErrorCode.PROGRAM_VALIDATION_FAILED,
   730	            errors=report.errors,
   731	            safety_status=safety_status,
   732	            rejected_candidates=rejected,
   733	            decision_trace=trace,
   734	        )
   735	    program, report = _attach_coach_quality_metrics(program, request, normalized, report, ruleset)
   736	    quality_metrics = build_coach_quality_metrics(program, request, report, ruleset)
   737	    final_trace = trace + (
   738	        {
   739	            "stage": "final_construction",
   740	            "status": "succeeded",
   741	            "reason_codes": ("FINAL_CONSTRUCTION_SUCCEEDED",),
   742	        },
   743	        {"stage": "coach_quality", "metrics": quality_metrics},
   744	    )
   745	    report = replace(report, decision_trace=final_trace)
   746	    program = replace(
   747	        program,
   748	        validation_report=report,
   749	        warnings=report.warnings,
   750	        decision_trace=final_trace,
   751	    )
   752	    return ProgramGenerationResult(
   753	        program=program,
   754	        safety_status=safety_status,
   755	        rejected_candidates=rejected,
   756	        decision_trace=final_trace,
   757	    )

   761	    request: ProgramGenerationRequest,
   762	    normalized: NormalizedProgramRequest,

   853	        {"stage": "normalization", "assumptions": normalized.assumptions},

   913	    unmet_priority_errors: list[str] = []
   914	    for muscle in sorted(explicit_priority_muscles, key=lambda item: item.value):
   915	        target = targets_by_muscle.get(muscle)
   916	        if target is None:
   917	            continue
   918	        direct_value = direct[muscle.value]
   919	        if target.direct_minimum_required and direct_value < target.minimum_direct_sets:
   920	            unmet_priority_errors.append(
   921	                f"TEMPLATE_PRIORITY_HARD_MINIMUM_UNSATISFIED:{muscle.value}"

   934	        f"TEMPLATE_PRIORITY_PREFERRED_TARGET_PARTIAL:{muscle.value}"
   935	        for muscle in sorted(
   936	            explicit_priority_muscles | body_priority_muscles,
   937	            key=lambda item: item.value,
   938	        )
   939	        if (target := targets_by_muscle.get(muscle))
   940	        and (
   941	            direct[muscle.value] < target.target_sets
   942	            or effective_volume.effective_sets_by_muscle.get(muscle.value, 0)
   943	            < target.effective_target_sets
   944	        )
   945	    )
   946	    if soft_priority_shortfalls:
   947	        trace += (
   948	            {
   949	                "stage": "template_priority_adaptation",
   950	                "status": "constrained",
   951	                "reason_codes": soft_priority_shortfalls,
   952	            },
   953	        )
   954	    days = add_cardio(normalized, days, cardio_eligible, ruleset)
   955	    program = WorkoutProgram(
   956	        user_profile_snapshot=request.model_dump(mode="json"),
   957	        engine_version=ruleset.engine_version,
   958	        ruleset_version=ruleset.version,
   959	        seed=normalized.seed,
   960	        primary_goal=normalized.primary_goal,
   961	        secondary_goal=request.secondary_goal_optional,
   962	        training_status=normalized.training_status,
   963	        safety_status=safety_status,
   964	        assumptions=normalized.assumptions,
   965	        warnings=(),
   966	        duration_weeks=request.program_duration_weeks,
   967	        split=split,
   968	        weekly_schedule=days,
   969	        progression_policy={
   970	            **double_progression_policy(ruleset),
   971	            "deload": deload_policy(ruleset),
   972	        },

   978	    report = validate_program(program, request, ruleset)
   979	    if not report.is_valid:
   980	        return ProgramGenerationResult(
   981	            program=None,
   982	            error_code=GenerationErrorCode.PROGRAM_VALIDATION_FAILED,

   984	            safety_status=safety_status,
   985	            rejected_candidates=rejected,
   986	            decision_trace=trace,
   987	        )
   988	    program, report = _attach_coach_quality_metrics(program, request, normalized, report, ruleset)
   989	    quality_metrics = build_coach_quality_metrics(program, request, report, ruleset)
   990	    final_trace = trace + (
   991	        {
   992	            "stage": "final_construction",
   993	            "status": "succeeded",
   994	            "reason_codes": ("FINAL_CONSTRUCTION_SUCCEEDED",),
   995	        },
   996	        {"stage": "coach_quality", "metrics": quality_metrics},
   997	    )
   998	    report = replace(report, decision_trace=final_trace)
   999	    return ProgramGenerationResult(
  1000	        program=replace(
  1001	            program,
  1002	            validation_report=report,
  1003	            warnings=report.warnings,
  1004	            decision_trace=final_trace,
  1005	        ),
  1006	        safety_status=safety_status,
  1007	        rejected_candidates=rejected,
  1008	        decision_trace=final_trace,
  1009	    )

  1051	def _duration_repair_trace(
  1052	    capacity: SessionCapacity,
  1053	    before: tuple[WorkoutDay, ...],
  1054	    after: tuple[WorkoutDay, ...],
  1055	    reason_codes: tuple[str, ...],
  1056	) -> dict[str, object]:
  1057	    repair_applied = "SESSION_DURATION_REPAIR_APPLIED" in reason_codes
  1058	    duration_deltas = tuple(
  1059	        abs(updated.estimated_duration_minutes - original.estimated_duration_minutes)
  1060	        for original, updated in zip(before, after, strict=True)
  1061	    )
  1062	    exercise_deltas = tuple(
  1063	        abs(len(updated.exercises) - len(original.exercises))
  1064	        for original, updated in zip(before, after, strict=True)
  1065	    )
  1066	    if not repair_applied:
  1067	        classification = "not_needed"
  1068	    elif max(duration_deltas, default=0) <= 10 and max(exercise_deltas, default=0) <= 1:
  1069	        classification = "minor"
  1070	    else:
  1071	        classification = "major"
  1072	    unavoidable_constraints = tuple(
  1073	        code for code in reason_codes if "CONSTRAINED" in code or "EXTENDED" in code
  1074	    )
  1075	    return {
  1076	        "stage": "session_duration",
  1077	        "status": "repaired" if repair_applied else "validated",
  1078	        "planned_resistance_work_budget_minutes": capacity.resistance_work_budget_minutes,
  1079	        "planned_exercise_capacity": capacity.expected_exercise_count_capacity,
  1080	        "planned_set_capacity": capacity.expected_working_set_capacity,
  1081	        "estimated_duration_before_late_repair": tuple(
  1082	            day.estimated_duration_minutes for day in before
  1083	        ),
  1084	        "estimated_duration_after_late_repair": tuple(
  1085	            day.estimated_duration_minutes for day in after
  1086	        ),
  1087	        "repair_classification": classification,
  1088	        "unavoidable_duration_constraints": unavoidable_constraints,
  1089	        "reason_codes": reason_codes,
  1090	    }

  1093	def _coach_quality_metrics(
  1094	    program: WorkoutProgram,
  1095	    request: ProgramGenerationRequest,
  1096	    normalized: NormalizedProgramRequest,
  1097	    report: ValidationReport,
  1098	    ruleset: ProgramRuleset,
  1099	) -> dict[str, object]:
  1100	    trace_reason_codes = {
  1101	        code
  1102	        for entry in program.decision_trace
  1103	        for key in ("reason_codes", "reasons")
  1104	        for code in _string_sequence(entry.get(key))
  1105	    }
  1106	    ranges = program.aggregate_metrics.get("volume_ranges_by_muscle", {})
  1107	    hard_volume_exceeded = isinstance(ranges, dict) and any(
  1108	        isinstance(values, dict)
  1109	        and float(values.get("actual_effective_volume", 0))
  1110	        > float(values.get("effective_maximum_hard", ruleset.maximum_sets[program.training_status]))
  1111	        for values in ranges.values()

  1371	        constraint_reasons.append("VOLUME_CONSTRAINED_BY_TEMPLATE_STRUCTURE")

  1376	    elif constraint_reasons:
  1377	        status = "constrained"

  1408	def _priority_metrics(
  1409	    days: tuple[WorkoutDay, ...],
  1410	    volume: WeeklyVolumePlan,
  1411	    direct_sets: dict[str, int],
  1412	    effective_sets: dict[str, float],
  1413	    policy: PriorityAllocationPolicy,
  1414	    ruleset: ProgramRuleset,
  1415	) -> dict[str, dict[str, object]]:
  1416	    metrics: dict[str, dict[str, object]] = {}
  1417	    for muscle in policy.priorities:
  1418	        session_indexes = tuple(
  1419	            day.day_index
  1420	            for day in days
  1421	            if any(item.primary_muscle is muscle for item in day.exercises)
  1422	        )
  1423	        direct = direct_sets.get(muscle.value, 0)
  1424	        effective = effective_sets.get(muscle.value, 0.0)
  1425	        target = volume.direct_sets_for(muscle)
  1426	        effective_target = volume.effective_target_for(muscle)
  1427	        target_available = target > 0 or effective_target > 0
  1428	        direct_satisfied = target_available and direct >= target
  1429	        effective_satisfied = target_available and effective >= effective_target
  1430	        useful_frequency = policy.useful_frequency(target, ruleset)
  1431	        frequency_satisfied = len(session_indexes) >= useful_frequency
  1432	        status = (
  1433	            "satisfied"
  1434	            if direct_satisfied and effective_satisfied and frequency_satisfied
  1435	            else "partial"
  1436	        )
  1437	        reason_codes: list[str] = []
  1438	        if direct_satisfied and effective_satisfied:
  1439	            reason_codes.append("PRIORITY_VOLUME_INCREASED")
  1440	        else:
  1441	            reason_codes.append("PRIORITY_TARGET_PARTIALLY_SATISFIED")
  1442	        if frequency_satisfied and useful_frequency > 1:
  1443	            reason_codes.append("PRIORITY_FREQUENCY_INCREASED")
  1444	        else:
  1445	            reason_codes.append("PRIORITY_TARGET_CONSTRAINED")
  1446	        metrics[muscle.value] = {
  1447	            "direct_sets": direct,
  1448	            "effective_sets": effective,
  1449	            "target_sets": target,
  1450	            "effective_target_sets": effective_target,
  1451	            "preferred_frequency": useful_frequency,
  1452	            "session_frequency": len(session_indexes),
  1453	            "session_indexes": session_indexes,
  1454	            "distributed": frequency_satisfied,
  1455	            "status": status,
  1456	            "reason_codes": tuple(dict.fromkeys(reason_codes)),
  1457	        }
  1458	    return metrics
